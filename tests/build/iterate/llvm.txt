define i32 @"example.com/iterate.Sum"(ptr %xs) {
entry:
  %v.total = alloca i32
  %v.i = alloca i64
  %v.x = alloca i32
  store i32 0, ptr %v.total
  store i64 0, ptr %v.i
  br label %for.cond.1
for.cond.1:
  %t1 = load i64, ptr %v.i
  %t2 = icmp ult i64 %t1, 3
  br i1 %t2, label %for.body.1, label %for.end.1
for.body.1:
  %t3 = getelementptr inbounds [3 x i32], ptr %xs, i64 0, i64 %t1
  %t4 = load i32, ptr %t3
  store i32 %t4, ptr %v.x
  %t5 = load i32, ptr %v.total
  %t6 = load i32, ptr %v.x
  %t7 = add i32 %t5, %t6
  store i32 %t7, ptr %v.total
  %t8 = load i64, ptr %v.i
  %t9 = icmp ugt i64 %t8, 2
  br i1 %t9, label %if.then.2, label %if.end.2
if.then.2:
  br label %for.end.1
if.end.2:
  br label %for.step.1
for.step.1:
  %t10 = load i64, ptr %v.i
  %t11 = add i64 %t10, 1
  store i64 %t11, ptr %v.i
  br label %for.cond.1
for.end.1:
  %t12 = load i32, ptr %v.total
  ret i32 %t12
}
